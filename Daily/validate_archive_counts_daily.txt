#!/usr/bin/env bash
# Archive validation v4: two positional paths; results and logs beside audit.
# Run with --help. Requires Bash and awk only, not Python.
set -euo pipefail
export LC_ALL=C

usage() {
    cat <<'HELP'
Validate archived text-file counts using the audit CSV year and filename.
Requires Bash and awk; no Python. Input/output are CSV, not XLSX.

Usage:
  bash validate_archive_counts.txt /path/to/TYPE.csv /data/Claims/SOURCE_PARENT
  bash validate_archive_counts.txt /home/kalkanti/Counts/Monthly/RECP/RECP.csv /data/Claims/CLAIM_RECP

Required inputs (in this order):
  1. Full audit CSV path including filename.
  2. Source parent folder containing the year subfolders.

Optional options (after the two paths):
  --header-lines N    Header lines to exclude from each source file (default 0).
  --trailer-lines N   Trailer lines to exclude from each source file (default 0).
  --help             Show this help.

No file-type parameter or source-folder mapping is required.
Result filenames use the audit CSV basename; all outputs stay beside the CSV.

Expected audit columns: A=Year, B=File Name, C=File Date, D=Record Count, E=Status.
Input must contain a header and exactly these five columns. F/G are added.
A=2005 and B=e38624 resolve to SOURCE_ROOT/2005/e38624.
The year must be four digits. No searching in other years or nested directories.
Filenames must match exactly, including extensions and case.

Counting: one physical line per record; blank lines included. A last line without
newline also counts. Plain uncompressed text files only; source CSV with quoted
embedded newlines needs a logical-record parser instead. Exclusions apply to all
files in this run. Run after copying finishes against stable files.

Output: existing A-E values, F=actual count, G=Match/Not-Matched. A separate
OUTPUT.log.csv contains year, full source path, difference, and error reason.
OUTPUT.run.log records run settings, summary, and counting/parser errors.
All outputs, logs, and temporary work are created in the audit CSV directory.
Errors leave F blank and set G to Not-Matched. Original audit is never changed.
Existing outputs are never overwritten. CSV does not preserve Excel formatting.
Exit codes: 0=all matched; 1=completed with Not-Matched rows; 2=fatal error.
HELP
}
fail() { printf 'ERROR: %s\n' "$*" >&2; exit 2; }
need_value() { [[ $# -ge 2 && -n "$2" ]] || fail "Missing value for $1"; }
if [[ ${1:-} == --help || ${1:-} == -h ]]; then usage; exit 0; fi
[[ $# -ge 2 ]] || { usage >&2; fail 'Supply the audit CSV path and source parent folder.'; }
input=$1
source_root=$2
shift 2
file_type=$(basename -- "$input")
file_type=${file_type%.*}
output=
export HEADER_LINES=0 TRAILER_LINES=0
while [[ $# -gt 0 ]]; do
    case "$1" in
        --help|-h) usage; exit 0 ;;
        --header-lines) need_value "$@"; HEADER_LINES=$2; shift 2 ;;
        --trailer-lines) need_value "$@"; TRAILER_LINES=$2; shift 2 ;;
        *) fail "Unknown argument: $1. Use --help." ;;
    esac
done
[[ -f "$input" && -r "$input" ]] || fail "Audit CSV missing or unreadable: $input"
case "$input" in
    *.[cC][sS][vV]) ;;
    *) fail 'Save the original Excel worksheet as CSV UTF-8, not just a renamed XLSX.' ;;
esac
[[ -d "$source_root" && -x "$source_root" ]] || fail "Source root missing or inaccessible: $source_root"
export AUDIT_SOURCE_DIR
AUDIT_SOURCE_DIR=$(cd -- "$source_root" && pwd -P)
audit_dir=$(cd -- "$(dirname -- "$input")" && pwd -P)
output=${output:-$audit_dir/${file_type}_validated_$(date +%Y%m%d_%H%M%S)_$$.csv}
case "$output" in *.[cC][sS][vV]) ;; *) fail 'Output must have a .csv extension.' ;; esac
# Keep all generated files in the audit directory.
if [[ "$output" != */* ]]; then output="$audit_dir/$output"; fi
[[ -d "$(dirname -- "$output")" ]] || fail 'Output directory does not exist.'
output_dir=$(cd -- "$(dirname -- "$output")" && pwd -P)
[[ "$output_dir" == "$audit_dir" ]] || fail 'Output must be in the same folder as the audit CSV.'
output="$audit_dir/$(basename -- "$output")"
logfile="${output}.log.csv"
runlog="${output}.run.log"
[[ -d "$(dirname -- "$output")" ]] || fail 'Output directory does not exist.'
if [[ -e "$output" || -L "$output" || -e "$logfile" || -L "$logfile" || -e "$runlog" || -L "$runlog" ]]; then
    fail 'Output or log already exists. Choose a new output filename.'
fi
for number in "$HEADER_LINES" "$TRAILER_LINES"; do
    [[ "$number" =~ ^[0-9]+$ ]] || fail 'Header/trailer counts must be nonnegative integers.'
done
printf 'Type: %s\nAudit: %s\nSource root: %s\nOutput: %s\n' "$file_type" "$input" "$AUDIT_SOURCE_DIR" "$output"
work_dir=$(mktemp -d "$audit_dir/.validation_work.XXXXXX")
trap 'rm -rf -- "$work_dir"' EXIT
export AUDIT_LOG_TMP="$work_dir/log.csv"
export AUDIT_SUMMARY_TMP="$work_dir/summary.txt"
awk_rc=0
(set -o noclobber; printf 'Type: %s\nAudit: %s\nSource: %s\nOutput: %s\nHeader lines: %s\nTrailer lines: %s\nStarted: %s\n' \
    "$file_type" "$input" "$AUDIT_SOURCE_DIR" "$output" "$HEADER_LINES" "$TRAILER_LINES" "$(date)" > "$runlog") || fail 'Could not create run log.'
awk '
function quote_csv(s) { gsub(/"/, "\"\"", s); return "\"" s "\"" }
function shell_quote(s,  q,out,i,c) {
    q=sprintf("%c",39); out=q
    for(i=1;i<=length(s);i++) {
        c=substr(s,i,1)
        out=out (c==q ? q "\"" q "\"" q : c)
    }
    return out q
}
# Returns zero for an incomplete quoted record, -1 for malformed CSV,
# or a positive field count. States: 0=start, 1=plain, 2=quoted, 3=closed.
function parse_csv(s, a,  n,state,i,c) {
    for(i in a) delete a[i]
    n=1; state=0; a[n]=""
    for(i=1;i<=length(s);i++) {
        c=substr(s,i,1)
        if(state==2) {
            if(c=="\"") {
                if(substr(s,i+1,1)=="\"") { a[n]=a[n] "\""; i++ }
                else state=3
            } else a[n]=a[n] c
        } else if(c==",") { n++; a[n]=""; state=0 }
        else if(state==3) return -1
        else if(c=="\"") { if(state!=0) return -1; state=2 }
        else { a[n]=a[n] c; state=1 }
    }
    return state==2 ? 0 : n
}
function fatal(message) {
    print "ERROR: " message > summary
    failed=1
    exit 2
}
BEGIN {
    root=ENVIRON["AUDIT_SOURCE_DIR"]
    logpath=ENVIRON["AUDIT_LOG_TMP"]; summary=ENVIRON["AUDIT_SUMMARY_TMP"]
    headers=ENVIRON["HEADER_LINES"]+0; trailers=ENVIRON["TRAILER_LINES"]+0
    print "Audit Row,Year,File Name,Expected Count,Actual Count,Status,Reason,Difference,Source Path" > logpath
}
{
    line=$0
    if(NR==1) sub(/^\357\273\277/, "", line)
    sub(/\r$/, "", line)
    if(!pending && line=="") next
    record=pending ? record "\n" line : line
    fields=parse_csv(record, a)
    if(fields==0) { pending=1; next }
    pending=0
    if(fields<0) fatal("Malformed CSV near physical line " NR)
    row++
    if(fields!=5) fatal("Audit row " row " must have exactly five columns A-E; save the original sheet as comma-delimited CSV")
    original=""
    for(i=1;i<=5;i++) original=original (i>1 ? "," : "") quote_csv(a[i])
    if(row==1) {
        print original ",Actual File Record Count,Count Validation"
        next
    }
    if(a[1]=="" && a[2]=="" && a[3]=="" && a[4]=="" && a[5]=="") {
        print original ",,"
        next
    }
    actual=""; status="Not-Matched"; reason=""; difference=""; path=""
    name=a[2]; expected=a[4]; year=a[1]
    gsub(/^[ \t]+|[ \t]+$/, "", year)
    gsub(/^[ \t]+|[ \t]+$/, "", expected)
    if(expected ~ /^[0-9][0-9]?[0-9]?(,[0-9][0-9][0-9])+$/) gsub(/,/, "", expected)
    if(year !~ /^[0-9][0-9][0-9][0-9]$/)
        reason="Invalid year in column A; expected four digits"
    else if(expected !~ /^[0-9]+$/ || expected+0>9007199254740991)
        reason="Invalid expected record count in column D"
    else if(name=="" || name=="." || name==".." || name ~ /[\/\r\n]/)
        reason="Column B must contain a complete filename without a directory"
    else if(tolower(name) ~ /\.(gz|zip|bz2|xz|z|tar|xlsx|xls|parquet|avro)$/)
        reason="Unsupported compressed or binary source; extract or use a format-specific parser"
    else {
        path=root "/" year "/" name
        if(!(path in cache_reason)) {
            quoted=shell_quote(path)
            command="test -f " quoted " && test -r " quoted " && test ! -L " quoted
            if(system("test -d " shell_quote(root "/" year) " && test -x " shell_quote(root "/" year))!=0)
                cache_reason[path]="Year directory missing or inaccessible"
            else if(system(command)!=0) cache_reason[path]="File missing, unreadable, nonregular, or symlink"
            else {
                total=0
                while((read_result=(getline source_line < path))>0) total++
                close(path)
                if(read_result<0) cache_reason[path]="Error reading source file"
                else if(total<headers+trailers) cache_reason[path]="Fewer lines than configured header plus trailer"
                else { cache_reason[path]="OK"; cache_count[path]=total-headers-trailers }
            }
        }
        if(cache_reason[path]!="OK") reason=cache_reason[path]
        else {
            actual=sprintf("%.0f",cache_count[path])
            difference=sprintf("%.0f",actual-expected)
            if(actual+0==expected+0) { status="Match"; reason="Counts equal" }
            else reason="Count difference"
        }
    }
    print original "," actual "," status
    print row "," quote_csv(a[1]) "," quote_csv(name) "," quote_csv(a[4]) "," actual "," status "," quote_csv(reason) "," difference "," quote_csv(path) > logpath
    checked++
    if(status=="Match") matched++; else mismatched++
}
END {
    if(failed) exit 2
    if(pending) { print "ERROR: Unclosed quoted CSV field" > summary; exit 2 }
    if(!row) { print "ERROR: Empty audit CSV" > summary; exit 2 }
    print "Rows: " checked+0 " | Match: " matched+0 " | Not-Matched: " mismatched+0 > summary
    if(mismatched) exit 1
}
' < "$input" > "$work_dir/output.csv" 2>> "$runlog" || awk_rc=$?

if [[ -f "$AUDIT_SUMMARY_TMP" ]]; then
    cat -- "$AUDIT_SUMMARY_TMP"
    cat -- "$AUDIT_SUMMARY_TMP" >> "$runlog"
fi
if [[ $awk_rc -gt 1 ]]; then
    printf 'Validation failed; see run log: %s\n' "$runlog" >&2
    printf 'Fatal validation error; no results CSV created.\n' >> "$runlog"
    exit 2
fi
# noclobber also prevents overwriting files created by another process mid-run.
if ! (set -o noclobber; cat -- "$work_dir/output.csv" > "$output"); then
    printf 'Could not create output: %s\n' "$output" >&2
    exit 2
fi
if ! (set -o noclobber; cat -- "$AUDIT_LOG_TMP" > "$logfile"); then
    printf 'Output created, but could not create log: %s\n' "$logfile" >&2
    exit 2
fi
printf 'Finished: %s\nExit status: %s\n' "$(date)" "$awk_rc" >> "$runlog"
printf 'Created: %s\nDetails: %s\nRun log: %s\n' "$output" "$logfile" "$runlog"
exit "$awk_rc"
